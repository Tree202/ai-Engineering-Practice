一、第 ① 层:单元测试

5 条 · 实测 0.00 秒 · 准备段为零

# 文件:tests/test_price.py
"""第 1 层:单元测试(金字塔最底下、数量最多、跑得最快的一层)。

这一层测什么:
    一个纯函数的输入输出,包括正常值和边界值(0、进位、非法输入)。
这一层不测什么:
    不碰数据库、不碰网络、不启动服务。
    如果一个「单元测试」需要连数据库,那它其实已经是集成测试了。
"""

import pytest

from myshop.price import format_price


def test_普通价格() -> None:
    assert format_price(1234) == "¥12.34"


def test_零元是边界值() -> None:
    # 0 很容易被漏掉,专门写一条
    assert format_price(0) == "¥0.00"


def test_不足一角要在前面补零() -> None:
    # 5 分不能显示成 "¥0.5",必须是 "¥0.05"
    assert format_price(5) == "¥0.05"


def test_整数元不能丢掉小数位() -> None:
    assert format_price(10000) == "¥100.00"


def test_负数要报错() -> None:
    # 「该报错的时候有没有报错」也是行为的一部分,同样要测
    with pytest.raises(ValueError):
        format_price(-1)
注意文件开头那段说明 每个测试文件开头都写清了「测什么 / 不测什么」。这不是形式主义 —— 半年后你回来加测试时,这段话决定了你会不会把一条集成测试塞进单元测试文件里。

五条测试的分工

测试输入守着什么
test_普通价格1234基本功能
test_零元是边界值00 是最常被漏掉的值
test_不足一角要在前面补零5补零逻辑
test_整数元不能丢掉小数位10000小数位不能被吞
test_负数要报错-1该拦的拦不拦得住
💡 第 8 页会证明:这五条各有各的不可替代性 五次改坏实验里,每一次立功的都是不同的测试。 留任何一条、砍掉其余,都会漏掉别的 bug。

二、第 ② 层:集成测试

5 条 · 实测 0.02 秒 · 准备段:要一个临时数据库

# 文件:tests/test_order.py
"""第 2 层:集成测试(金字塔中间层)。

这一层测什么:
    多个模块拼在一起还对不对 —— order 调用 price 有没有接错,
    数据有没有真的写进数据库、换个连接还能不能读回来。
这一层不测什么:
    不测 format_price 的每个边界(那是第 1 层的活,重复测是浪费)。
    不走网络,直接调 Python 函数。

tmp_path 是运行器自带的准备工作:每个测试自动分到一个全新的空目录,
跑完自动清理。所以测试之间不会互相污染数据库。
"""

from pathlib import Path

import pytest

from myshop.order import create_order, get_order


def test_下单会把总价算对并且带上价格文案(tmp_path: Path) -> None:
    数据库 = str(tmp_path / "test.db")

    订单 = create_order("咖啡", unit_cents=1250, qty=2, db_path=数据库)

    assert 订单["total_cents"] == 2500
    # 这一行证明 order 模块真的用了 price 模块,而不是自己乱拼字符串
    assert 订单["total_text"] == "¥25.00"


def test_订单真的落盘了_换个连接还读得到(tmp_path: Path) -> None:
    数据库 = str(tmp_path / "test.db")

    订单 = create_order("键盘", unit_cents=39900, qty=1, db_path=数据库)
    # get_order 会重新连接一次数据库,读到说明数据不是只存在内存里
    读回来 = get_order(订单["id"], db_path=数据库)

    assert 读回来["item"] == "键盘"
    assert 读回来["total_cents"] == 39900
    assert 读回来["total_text"] == "¥399.00"


def test_下两单会拿到两个不同的订单号(tmp_path: Path) -> None:
    数据库 = str(tmp_path / "test.db")

    第一单 = create_order("茶", unit_cents=800, qty=1, db_path=数据库)
    第二单 = create_order("茶", unit_cents=800, qty=1, db_path=数据库)

    # 自增主键有没有正常工作,是数据库层面的行为,只能靠集成测试发现
    assert 第一单["id"] != 第二单["id"]


def test_数量为零应该被拦下来并且不写库(tmp_path: Path) -> None:
    数据库 = str(tmp_path / "test.db")

    with pytest.raises(ValueError):
        create_order("咖啡", unit_cents=1250, qty=0, db_path=数据库)

    # 报错之后数据库里不该留下垃圾数据
    with pytest.raises(LookupError):
        get_order(1, db_path=数据库)


def test_单价为负数也要被拦下来并且不写库(tmp_path: Path) -> None:
    """这条测的是「校验顺序」这个容易写错的地方。

    如果 create_order 先写库再算 format_price,那么负数单价会
    「先把脏数据写进库,然后才抛异常」—— 库里就留下了一条不该存在的订单。
    校验必须全部发生在写库之前。
    """
    数据库 = str(tmp_path / "test.db")

    with pytest.raises(ValueError):
        create_order("退款商品", unit_cents=-100, qty=1, db_path=数据库)

    # 关键断言:异常抛出之后,库里必须一条记录都没有
    with pytest.raises(LookupError):
        get_order(1, db_path=数据库)

这一层的三个关键设计

设计代码为什么
① 测试隔离 tmp_path 参数 每条测试拿到不同的空目录,各写各的数据库文件。跑完自动清理
② 证明跨模块 assert 订单["total_text"] == "¥25.00" 只断言 total_cents == 2500 是不够的 —— 那只证明乘法对了。 这个字符串只能来自另一个模块
③ 证明落盘 get_order(...) 重新连接 同一个连接里没提交的数据照样能读到。换个连接读到,才算真的写进文件了
试试看:故意把它弄坏 把 order.py 里 format_price(total_cents) 那一行删掉,改成自己拼字符串。
看 assert 订单["total_text"] == "¥25.00" 这条变红。

亲眼看到测试抓住一个真实的错误,比读十页文档管用。

三、第 ③ 层:接口测试

7 条 · 实测 3.64 秒 · 准备段:要真的启动一个服务器

这一层的「准备」重到必须单独写一个函数:

# 文件:tests/test_api.py (节选)——完整代码见配套 myshop 项目

@pytest.fixture()                                    # ← 声明这是一个「准备工作」
def 服务器(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> Iterator[str]:
    """启动一个真的服务器,测完自动关掉。返回它的地址。"""
    # 让接口层写到临时数据库,别碰真实的数据文件
    monkeypatch.setattr(order, "DB_PATH", str(tmp_path / "api.db"))

    服务 = api.make_server(port=0)   # 0 = 让系统分配空闲端口,避免端口冲突
    # daemon=True:主进程结束时这个线程不会拖住不放
    线程 = threading.Thread(target=服务.serve_forever, daemon=True)
    线程.start()

    yield f"http://127.0.0.1:{服务.server_port}"

    # yield 之后的代码是「收尾」,不管测试成功失败都会执行
    服务.shutdown()
    服务.server_close()
    线程.join(timeout=5)


def 发请求(url, method, payload=None, raw_body=None):
    """一个小助手:发一个请求,返回 (状态码, 解析好的数据)。"""
    if raw_body is not None:
        data = raw_body
    elif payload is not None:
        data = json.dumps(payload).encode("utf-8")
    else:
        data = None

    请求 = urllib.request.Request(
        url, data=data,
        headers={"Content-Type": "application/json"},
        method=method,
    )
    try:
        with urllib.request.urlopen(请求) as 响应:
            return 响应.status, json.loads(响应.read())
    except urllib.error.HTTPError as 错误:
        # 4xx/5xx 是以异常形式抛出来的,要接住才能断言状态码
        return 错误.code, json.loads(错误.read())


def test_下单接口返回_201_和订单内容(服务器: str) -> None:
    状态码, 数据 = 发请求(
        f"{服务器}/orders", "POST",
        {"item": "咖啡", "unit_cents": 1250, "qty": 2},
    )

    assert 状态码 == 201            # 201 = 新建资源成功
    assert 数据["item"] == "咖啡"
    assert 数据["total_cents"] == 2500
    assert 数据["total_text"] == "¥25.00"


def test_请求体不是合法_JSON_要返回_400_而不是_500(服务器: str) -> None:
    """这条测的是「解析放在 try 里面还是外面」这个容易写错的地方。

    如果解析写在 try 外面,一个畸形请求会让服务器抛未捕获异常,
    客户端拿到 500(服务器内部错误),终端还会打一屏错误。
    正确的行为是老老实实返回 400(是你请求发错了,不是我崩了)。
    """
    # 注意:字节串字面量 b"..." 里只能放 ASCII 字符,
    # 想发中文得先写成普通字符串再 .encode("utf-8")
    畸形请求体 = "{这不是合法的 JSON".encode("utf-8")

    状态码, 数据 = 发请求(f"{服务器}/orders", "POST", raw_body=畸形请求体)

    assert 状态码 == 400
    assert "error" in 数据

四个实用技巧

技巧说明
随机端口 port=0 让系统挑一个空闲的。写死 8000 的话,你本地开着开发服务器时测试就跑不了
临时替换变量 monkeypatch.setattr 把数据库路径临时指向临时目录,测试结束自动改回来
4xx 是异常 标准库的网络模块遇到 4xx/5xx 会抛异常而不是返回响应, 必须接住才能断言状态码。新手在这里常以为「接口挂了」
字节串只能 ASCII b"中文" 是语法错误。要先写普通字符串再 .encode("utf-8")
⚠ 最后那条是我自己踩过的 写这个项目时,我第一版就写了 b"{这不是合法的 JSON"。 三个工具在同一处同时报错:
  真实输出
ruff:  invalid-syntax: bytes can only contain ASCII literal characters
mypy:  error: Bytes can only contain ASCII literal characters.  [syntax]
pytest: SyntaxError: bytes can only contain ASCII literal characters

这就是「让它自测」在真实场景里的样子 —— 三道检查同时拦下,我修完才能继续。

四、为什么接口层慢 180 倍

答案出人意料:慢的不是网络 实测:7 条接口测试 3.64 秒,5 条集成测试只要 0.02 秒(本项目本机实测值,不可外推到其他项目)。

慢的是每条测试都要「启动和关闭一个服务器」 —— 而关闭操作内部有一个固定的等待间隔。7 条测试就是 7 次启动 + 7 次关闭。
这个数字比任何说教都能解释「为什么金字塔上窄下宽」 不是接口测试写得不好,是「搭场景」本身就贵。

所以正确的做法不是「少写接口测试」,而是「把能在下层测的东西放到下层去测」 —— 价格进位这种细节,在单元层测一次 0.000 秒,在接口层测一次要 0.5 秒。

五、第 ④ 层:E2E

两个值得学的写法

写法为什么
导不到就跳过 没装浏览器工具的人跑测试时,输出是干净的 1 skipped 而不是一屏错误。 这样别人复制你的项目也不会看到一片红。 注意要守两道:只守 playwright 不守 pytest-playwright 的话,模块导入成功、 测试却会报 fixture not found 而不是跳过
按「看得见的东西」定位 用「按钮上的文字」而不是内部类名。类名改个字测试就挂,但按钮上的字改了才是真的改了功能。
🔄 经核实修正:那个「自动等待」的官方叫法 很多教程管它叫「web-first assertions」。官方在动作侧的术语是 auto-waiting,有专门文档页(/python/docs/actionability);而「web-first assertions」并没被废弃 —— 它今天仍出现在官方 Python 文档的 Handles 页与 v1.18 发行说明里。两个词指的不是同一件事:auto-waiting 是动作前会等元素可操作,web-first assertions 是断言会重试。

官方用词是:动作侧叫「自动等待(auto-waiting)」,有专门的文档页; 断言侧的表述是「断言会等待直到条件满足」,默认超时 5 秒。

行为描述是对的 —— 只是术语要改。顺带记住: page.wait_for_timeout 和 time.sleep 才是官方明确劝退的「硬等待」。
